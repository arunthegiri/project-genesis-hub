PART 1 — DATABASE MIGRATION
================================================================================

Create file:
    src/main/resources/db/migration/V3__strategies.sql

Content:

    -- Strategy definitions table
    CREATE TABLE IF NOT EXISTS strategies (
        id          BIGSERIAL PRIMARY KEY,
        name        VARCHAR(100) NOT NULL UNIQUE,
        description TEXT,
        definition  JSONB NOT NULL,
        created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
        updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    -- Backtest results table
    CREATE TABLE IF NOT EXISTS backtest_results (
        id              BIGSERIAL PRIMARY KEY,
        strategy_id     BIGINT REFERENCES strategies(id) ON DELETE CASCADE,
        symbol          VARCHAR(20),
        from_ts         TIMESTAMPTZ,
        to_ts           TIMESTAMPTZ,
        interval        VARCHAR(20) NOT NULL DEFAULT '1Min',
        total_trades    INT,
        winning_trades  INT,
        losing_trades   INT,
        win_rate        NUMERIC(8,4),
        total_pnl       NUMERIC(18,4),
        total_pnl_pct   NUMERIC(8,4),
        avg_win         NUMERIC(18,4),
        avg_loss        NUMERIC(18,4),
        largest_win     NUMERIC(18,4),
        largest_loss    NUMERIC(18,4),
        profit_factor   NUMERIC(8,4),
        max_drawdown    NUMERIC(8,4),
        sharpe_ratio    NUMERIC(8,4),
        trades          JSONB,
        equity_curve    JSONB,
        created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
    );

    -- Index for fast strategy lookup by name
    CREATE INDEX IF NOT EXISTS idx_strategies_name ON strategies(name);

    -- Index for fast backtest result lookup by strategy
    CREATE INDEX IF NOT EXISTS idx_backtest_strategy_id
        ON backtest_results(strategy_id);

Create: src/main/java/com/stocktracker/model/Strategy.java

    @Entity
    @Table(name = "strategies")
    @Data @Builder @NoArgsConstructor @AllArgsConstructor
    public class Strategy {
        @Id
        @GeneratedValue(strategy = GenerationType.IDENTITY)
        private Long id;

        @Column(nullable = false, unique = true)
        private String name;

        private String description;

        @Column(columnDefinition = "jsonb", nullable = false)
        private String definition;

        @Column(name = "created_at", nullable = false, updatable = false)
        @Builder.Default
        private Instant createdAt = Instant.now();

        @Column(name = "updated_at", nullable = false)
        @Builder.Default
        private Instant updatedAt = Instant.now();
    }

Create: src/main/java/com/stocktracker/model/BacktestResult.java

    @Entity
    @Table(name = "backtest_results")
    @Data @Builder @NoArgsConstructor @AllArgsConstructor
    public class BacktestResult {
        @Id
        @GeneratedValue(strategy = GenerationType.IDENTITY)
        private Long id;

        @ManyToOne(fetch = FetchType.LAZY)
        @JoinColumn(name = "strategy_id")
        private Strategy strategy;

        private String symbol;
        private Instant fromTs;
        private Instant toTs;
        private String interval;
        private Integer totalTrades;
        private Integer winningTrades;
        private Integer losingTrades;
        private BigDecimal winRate;
        private BigDecimal totalPnl;
        private BigDecimal totalPnlPct;
        private BigDecimal avgWin;
        private BigDecimal avgLoss;
        private BigDecimal largestWin;
        private BigDecimal largestLoss;
        private BigDecimal profitFactor;
        private BigDecimal maxDrawdown;
        private BigDecimal sharpeRatio;

        @Column(columnDefinition = "jsonb")
        private String trades;

        @Column(columnDefinition = "jsonb")
        private String equityCurve;

        @Column(name = "created_at", nullable = false, updatable = false)
        @Builder.Default
        private Instant createdAt = Instant.now();
    }

PART 5 — SERVICE
================================================================================

Create: src/main/java/com/stocktracker/service/StrategyService.java

    Methods:

    saveStrategy(StrategyRequest request) -> StrategyResponse
        - If strategy with same name exists, update it (upsert)
        - Save definition as JSON string
        - Save backtest results to backtest_results table
        - Return StrategyResponse

    getAllStrategies() -> List<StrategyResponse>
        - Return all strategies ordered by created_at desc

    getStrategy(String name) -> StrategyDetailResponse
        - Find strategy by name
        - Attach latest backtest result
        - Throw 404 if not found

    deleteStrategy(String name)
        - Delete strategy and all associated backtest results (cascade)
        - Throw 404 if not found

    getBacktestResults(String name) -> List<BacktestResultResponse>
        - Return all backtest runs for a given strategy name

PART 6 — CONTROLLER
================================================================================

Create: src/main/java/com/stocktracker/controller/StrategyController.java

    @RestController
    @RequestMapping("/api/strategies")
    public class StrategyController {

        POST /api/strategies
            @PostMapping
            Body: StrategyRequest
            Returns: 201 Created with StrategyResponse
            Description: Save a strategy and its backtest results from Jupyter

        GET /api/strategies
            @GetMapping
            Returns: PagedResponse<StrategyResponse>
            Description: List all saved strategies for the dashboard dropdown

        GET /api/strategies/{name}
            @GetMapping("/{name}")
            Returns: StrategyDetailResponse
            Description: Get full strategy with latest backtest results

        DELETE /api/strategies/{name}
            @DeleteMapping("/{name}")
            Returns: 204 No Content
            Description: Remove a strategy

        GET /api/strategies/{name}/results
            @GetMapping("/{name}/results")
            Returns: List<BacktestResultResponse>
            Description: Get all backtest runs for a strategy
    }
